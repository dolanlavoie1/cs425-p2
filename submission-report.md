## Results

| Window | Loss | Mean Time | Mean Throughput |
| ------ | ---- | ---- | ---------- |
| 1      | 0    | 106.374s | 9.63 KiB/s |
| 1      | 0.05 | 135.677s | 7.58 KiB/s |
| 16     | 0    | 6.801s | 150.57 KiB/s |
| 16     | 0.05 | 21.511s | 47.6 KiB/s |

1. In the window 1 loss 0 run, 1025 packets were sent at 100ms round trip time each. This comes out to 102.5 seconds of round trip time, with about 3.9 seconds of time where we weren't waiting for packets to be sent. That time is made up of the server starting up, the server waiting a few seconds to close after the FIN, and other overhead from python, the c programs, and possibly WSL. 
2. The speedup between a window of 1 and a window of 16 with a loss of 0 is about a 16x speed improvement because it is able to send 16 packets concurrently, while a window of 1 has to wait the entire 100ms between packets before it can send another. 
3. The 0.05 loss hurts a window of 16 more than a window of 1 because when a packet is lost, all packets above it have to be re-transmited. This means that for a packet of 1, if the packet is lost only that one packet has to be resent. However, in a window of 16, if a packet is lost then up to 16 packets might have to be sent again.

uint16_t lab_checksum(const uint8_t *data, size_t length)
{
    uint32_t sum = 0;

    for (size_t i = 0; i + 1U < length; i += 2U) {
        sum += ((uint32_t) data[i] << 8) | data[i + 1U];
        sum = (sum & UINT16_MAX) + (sum >> 16);
    }
    if ((length % 2U) != 0U) {
        sum += (uint32_t) data[length - 1U] << 8;
    }
    while (sum > UINT16_MAX) {
        sum = (sum & UINT16_MAX) + (sum >> 16);
    }
    return (uint16_t) ~sum;
}

bool lab_encode_packet(const LabPacket *packet, uint8_t *buffer,
                       size_t capacity, size_t *encoded_length)
{
    size_t total;
    uint16_t checksum;

    if (packet == NULL || buffer == NULL || encoded_length == NULL ||
        packet->type > LAB_FIN || packet->length > LAB_MAX_PAYLOAD ||
        ((packet->type == LAB_ACK || packet->type == LAB_FIN) &&
         packet->length != 0U)) {
        return false;
    }
    total = LAB_PACKET_HEADER + packet->length;
    if (capacity < total) {
        return false;
    }
    memset(buffer, 0, total);
    buffer[0] = (uint8_t) packet->type;
    buffer[4] = (uint8_t) (packet->seq >> 24);
    buffer[5] = (uint8_t) (packet->seq >> 16);
    buffer[6] = (uint8_t) (packet->seq >> 8);
    buffer[7] = (uint8_t) packet->seq;
    buffer[8] = (uint8_t) (packet->length >> 8);
    buffer[9] = (uint8_t) packet->length;
    if (packet->length > 0U) {
        memcpy(buffer + LAB_PACKET_HEADER, packet->payload, packet->length);
    }
    checksum = lab_checksum(buffer, total);
    buffer[2] = (uint8_t) (checksum >> 8);
    buffer[3] = (uint8_t) checksum;
    *encoded_length = total;
    return true;
}

bool lab_decode_packet(const uint8_t *buffer, size_t length, LabPacket *packet)
{
    uint16_t payload_length;

    if (buffer == NULL || packet == NULL || length < LAB_PACKET_HEADER ||
        length > LAB_MAX_PACKET || buffer[0] > LAB_FIN || buffer[1] != 0U) {
        return false;
    }
    payload_length = ((uint16_t) buffer[8] << 8) | buffer[9];
    if (payload_length > LAB_MAX_PAYLOAD ||
        LAB_PACKET_HEADER + (size_t) payload_length != length ||
        lab_checksum(buffer, length) != 0U) {
        return false;
    }
    if ((buffer[0] == LAB_ACK || buffer[0] == LAB_FIN) &&
        payload_length != 0U) {
        return false;
    }
    packet->type = (LabPacketType) buffer[0];
    packet->seq = ((uint32_t) buffer[4] << 24) |
                 ((uint32_t) buffer[5] << 16) |
                 ((uint32_t) buffer[6] << 8) | buffer[7];
    packet->length = payload_length;
    if (payload_length > 0U) {
        memcpy(packet->payload, buffer + LAB_PACKET_HEADER, payload_length);
    }
    return true;
}

static LabPacket sender_packet(const LabSender *sender, uint32_t sequence)
{
    LabPacket packet = {0};
    size_t offset;
    size_t remaining;

    packet.seq = sequence;
    if (sequence < sender->data_packets) {
        packet.type = LAB_DATA;
        offset = (size_t) sequence * LAB_MAX_PAYLOAD;
        remaining = sender->size - offset;
        packet.length = (uint16_t) (remaining > LAB_MAX_PAYLOAD ?
                                    LAB_MAX_PAYLOAD : remaining);
        if (packet.length > 0U) {
            memcpy(packet.payload, sender->data + offset, packet.length);
        }
    } else {
        packet.type = LAB_FIN;
    }
    return packet;
}

static size_t sender_emit(LabSender *sender, LabAction *actions, size_t capacity)
{
    size_t count = 0;

    while (sender->next < sender->base + sender->window &&
           sender->next < sender->data_packets && count < capacity) {
        actions[count].type = LAB_ACTION_SEND;
        actions[count].packet = sender_packet(sender, sender->next);
        count++;
        sender->next++;
    }
    if (sender->base == sender->data_packets &&
        sender->next == sender->base && count < capacity) {
        actions[count].type = LAB_ACTION_SEND;
        actions[count].packet = sender_packet(sender, sender->next);
        count++;
        sender->next++;
    }
    return count;
}

static void sender_arm(LabSender *sender, uint64_t now_ms)
{
    sender->timer_due_ms = now_ms + sender->timeout_ms;
    sender->timer_active = sender->base < sender->next;
}

void lab_sender_init(LabSender *sender, const uint8_t *data, size_t size,
                     uint32_t window, uint32_t timeout_ms)
{
    memset(sender, 0, sizeof(*sender));
    sender->data = data;
    sender->size = size;
    sender->data_packets = (uint32_t) ((size + LAB_MAX_PAYLOAD - 1U) /
                                       LAB_MAX_PAYLOAD);
    sender->window = window < 1U ? 1U :
                     (window > LAB_MAX_WINDOW ? LAB_MAX_WINDOW : window);
    sender->timeout_ms = timeout_ms == 0U ? 1U : timeout_ms;
}

size_t lab_sender_start(LabSender *sender, uint64_t now_ms,
                        LabAction *actions, size_t capacity)
{
    size_t count = sender_emit(sender, actions, capacity);

    if (count > 0U) {
        sender_arm(sender, now_ms);
    }
    return count;
}

size_t lab_sender_on_ack(LabSender *sender, uint32_t ack_seq,
                         uint64_t now_ms, LabAction *actions, size_t capacity)
{
    size_t count;

    if (ack_seq <= sender->base || ack_seq > sender->next) {
        return 0;
    }
    sender->base = ack_seq;
    sender->consecutive_timeouts = 0;
    if (sender->base == sender->next) {
        if (sender->base == sender->data_packets) {
            count = sender_emit(sender, actions, capacity);
            sender_arm(sender, now_ms);
            return count;
        }
        sender->timer_active = false;
        sender->complete = sender->base > sender->data_packets;
        return 0;
    }
    count = sender_emit(sender, actions, capacity);
    sender_arm(sender, now_ms);
    return count;
}

size_t lab_sender_on_timeout(LabSender *sender, uint64_t now_ms,
                             LabAction *actions, size_t capacity)
{
    size_t count = 0;

    if (!sender->timer_active) {
        return 0;
    }
    if (sender->consecutive_timeouts >= LAB_MAX_TIMEOUTS) {
        sender->failed = true;
        sender->timer_active = false;
        return 0;
    }
    for (uint32_t sequence = sender->base;
         sequence < sender->next && count < capacity; sequence++) {
        actions[count].type = LAB_ACTION_SEND;
        actions[count].packet = sender_packet(sender, sequence);
        count++;
    }
    sender->consecutive_timeouts++;
    sender_arm(sender, now_ms);
    return count;
}

size_t lab_receiver_on_packet(LabReceiver *receiver, const LabPacket *packet,
                              uint64_t now_ms, LabAction *actions,
                              size_t capacity)
{
    size_t count = 0;
    bool expected_fin;

    if (receiver->complete && packet->type == LAB_FIN &&
        packet->seq + 1U == receiver->expected && now_ms < receiver->linger_until_ms &&
        capacity > 0U) {
        actions[0].type = LAB_ACTION_SEND;
        memset(&actions[0].packet, 0, sizeof(actions[0].packet));
        actions[0].packet.type = LAB_ACK;
        actions[0].packet.seq = receiver->expected;
        return 1;
    }
    if (receiver->complete || receiver->failed || capacity == 0U) {
        return 0;
    }
    expected_fin = packet->type == LAB_FIN && packet->seq == receiver->expected;
    if (packet->type == LAB_DATA && packet->seq == receiver->expected) {
        actions[count].type = LAB_ACTION_DELIVER;
        actions[count].packet = *packet;
        count++;
        receiver->expected++;
    } else if (!expected_fin && packet->type != LAB_DATA) {
        return 0;
    } else if (!expected_fin) {
        /* A duplicate or out-of-order DATA still receives the cumulative ACK. */
    } else {
        receiver->expected++;
        receiver->complete = true;
        receiver->linger_until_ms = now_ms + LAB_RECEIVER_LINGER_MS;
    }
    if (count < capacity) {
        actions[count].type = LAB_ACTION_SEND;
        memset(&actions[count].packet, 0, sizeof(actions[count].packet));
        actions[count].packet.type = LAB_ACK;
        actions[count].packet.seq = receiver->expected;
        count++;
    }
    return count;
}

static bool relay_hello(int socket_fd, const struct addrinfo *relay,
                        const char *message)
{
    uint8_t response[128];
    struct pollfd descriptor = {.fd = socket_fd, .events = POLLIN};

    for (unsigned int attempt = 0; attempt < LAB_HELLO_ATTEMPTS; attempt++) {
        ssize_t sent = sendto(socket_fd, message, strlen(message), 0,
                              relay->ai_addr, relay->ai_addrlen);
        // GCOVR_EXCL_START
        if (sent != (ssize_t) strlen(message)) {
            continue;
        } // GCOVR_EXCL_STOP
        // GCOVR_EXCL_START
        if (poll(&descriptor, 1, LAB_HELLO_TIMEOUT_MS) <= 0) {
            continue;
        } // GCOVR_EXCL_STOP
        ssize_t received = recvfrom(socket_fd, response, sizeof(response) - 1U,
                                    0, NULL, NULL);
        // GCOVR_EXCL_START
        if (received < 0) {
            continue;
        } // GCOVR_EXCL_STOP
        response[received] = '\0';
        if (strcmp((const char *) response, "OK") == 0) {
            return true;
        }
        if (strncmp((const char *) response, "ERR ", 4) == 0) {
            fprintf(stderr, "%s\n", response);
            return false;
        }
    }
    fprintf(stderr, "relay registration timed out\n"); // GCOVR_EXCL_LINE
    return false; // GCOVR_EXCL_LINE
}

static bool load_file(const char *name, uint8_t **data, size_t *size)
{
    FILE *file = fopen(name, "rb");
    long length;
    uint8_t *contents;

    // GCOVR_EXCL_START
    if (file == NULL || fseek(file, 0, SEEK_END) != 0) {
        if (file != NULL) {
            (void) fclose(file);
        }
        return false;
    } // GCOVR_EXCL_STOP
    length = ftell(file);
    // GCOVR_EXCL_START
    if (length < 0 || fseek(file, 0, SEEK_SET) != 0) {
        (void) fclose(file);
        return false;
    } // GCOVR_EXCL_STOP
    contents = length == 0 ? NULL : malloc((size_t) length);
    // GCOVR_EXCL_START
    if (length > 0 && contents == NULL) {
        (void) fclose(file);
        return false;
    } // GCOVR_EXCL_STOP
    // GCOVR_EXCL_START
    if (length > 0 &&
        fread(contents, 1, (size_t) length, file) != (size_t) length) {
        free(contents);
        (void) fclose(file);
        return false;
    } // GCOVR_EXCL_STOP
    (void) fclose(file);
    *data = contents;
    *size = (size_t) length;
    return true;
}

        (void) snprintf(message, sizeof(message), "HELLO %s send %g %g %g",
                        options->session, options->loss, options->corrupt,
                        options->duplicate);
        // GCOVR_EXCL_START
        if (!relay_hello(socket_fd, address, message)) {
            free(data);
            (void) close(socket_fd);
            return 2;
        } // GCOVR_EXCL_STOP
    }
    lab_sender_init(&sender, data, size, options->window, options->timeout_ms);
    size_t count = lab_sender_start(&sender, wall_clock_ms(), actions,
                                    sizeof(actions) / sizeof(actions[0]));
    for (size_t i = 0; i < count; i++) {
        // GCOVR_EXCL_START
        if (!send_action(socket_fd, address, &actions[i])) {
            goto fail;
        } // GCOVR_EXCL_STOP
    }
    descriptor.fd = socket_fd;
    descriptor.events = POLLIN;
    while (!sender.complete && !sender.failed) {
        uint64_t current = wall_clock_ms();
        int timeout = sender.timer_active && sender.timer_due_ms > current ?
                      (int) (sender.timer_due_ms - current) : 0;
        int ready = poll(&descriptor, 1, timeout);
        // GCOVR_EXCL_START
        if (ready < 0) {
            if (errno == EINTR) {
                continue;
            }
            goto fail;
        } // GCOVR_EXCL_STOP
        // GCOVR_EXCL_START
        if (ready == 0) {
            count = lab_sender_on_timeout(&sender, wall_clock_ms(), actions,
                                          sizeof(actions) / sizeof(actions[0]));
        } // GCOVR_EXCL_STOP
        else {
            ssize_t length = recvfrom(socket_fd, received, sizeof(received),
                                      0, NULL, NULL);
            LabPacket packet;
            if (length < 0 || // GCOVR_EXCL_LINE
                !lab_decode_packet(received, (size_t) length, &packet) ||
                packet.type != LAB_ACK) {
                continue;
            }
            count = lab_sender_on_ack(&sender, packet.seq, wall_clock_ms(),
                                      actions, sizeof(actions) / sizeof(actions[0]));
        }
        for (size_t i = 0; i < count; i++) {
            // GCOVR_EXCL_START
            if (!send_action(socket_fd, address, &actions[i])) {
                goto fail;
            } // GCOVR_EXCL_STOP
        }
    }
    free(data);
    (void) close(socket_fd);
    return sender.complete ? 0 : 2;

// GCOVR_EXCL_START
fail:
    free(data);
    (void) close(socket_fd);
    return 2;
// GCOVR_EXCL_STOP
}

static int run_receiver(const LabTransferOptions *options,
                        const struct addrinfo *relay)
{
    const struct addrinfo *address;
    int socket_fd = open_udp(relay, &address);
    FILE *file;
    LabReceiver receiver;
    LabAction actions[2];
    uint8_t buffer[LAB_MAX_PACKET];
    struct pollfd descriptor;
    uint64_t last = wall_clock_ms();

    // GCOVR_EXCL_START
    if (socket_fd < 0) {
        return 2;
    } // GCOVR_EXCL_STOP
    file = fopen(options->file, "wb");
    // GCOVR_EXCL_START
    if (file == NULL) {
        (void) close(socket_fd);
        return 2;
    } // GCOVR_EXCL_STOP
    {
        char message[64];
        (void) snprintf(message, sizeof(message), "HELLO %s recv",
                        options->session);
        // GCOVR_EXCL_START
        if (!relay_hello(socket_fd, address, message)) {
            goto fail;
        } // GCOVR_EXCL_STOP
    }
    lab_receiver_init(&receiver);
    descriptor.fd = socket_fd;
    descriptor.events = POLLIN;
    while (!lab_receiver_expired(&receiver, wall_clock_ms()) &&
           wall_clock_ms() - last < LAB_RECEIVER_IDLE_MS) {
        uint64_t current = wall_clock_ms();
        int timeout = (int) (LAB_RECEIVER_IDLE_MS - (current - last));
        if (receiver.complete && receiver.linger_until_ms > current &&
            receiver.linger_until_ms - current < (uint64_t) timeout) {
            timeout = (int) (receiver.linger_until_ms - current);
        }
        int ready = poll(&descriptor, 1, timeout);
        // GCOVR_EXCL_START
        if (ready < 0) {
            if (errno == EINTR) {
                continue;
            }
            goto fail;
        } // GCOVR_EXCL_STOP
        // GCOVR_EXCL_START
        if (ready == 0) {
            if (receiver.complete &&
                wall_clock_ms() >= receiver.linger_until_ms) {
                break;
            }
            goto fail;
        } // GCOVR_EXCL_STOP
        ssize_t length = recvfrom(socket_fd, buffer, sizeof(buffer), 0, NULL, NULL);
        LabPacket packet;
        if (length < 0 || // GCOVR_EXCL_LINE
            !lab_decode_packet(buffer, (size_t) length, &packet)) {
            continue;
        }
        last = wall_clock_ms();
        size_t count = lab_receiver_on_packet(&receiver, &packet, last,
                                              actions, 2);
        for (size_t i = 0; i < count; i++) {
            // GCOVR_EXCL_START
            if (actions[i].type == LAB_ACTION_DELIVER &&
                fwrite(actions[i].packet.payload, 1, actions[i].packet.length,
                       file) != actions[i].packet.length) {
                goto fail;
            } // GCOVR_EXCL_STOP
            // GCOVR_EXCL_START
            if (actions[i].type == LAB_ACTION_SEND &&
                !send_action(socket_fd, address, &actions[i])) {
                goto fail;
            } // GCOVR_EXCL_STOP
        }
    }
    // GCOVR_EXCL_START
    if (!receiver.complete) {
        goto fail;
    } // GCOVR_EXCL_STOP
    (void) fclose(file);
    (void) close(socket_fd);
    return 0;

// GCOVR_EXCL_START
fail:
    (void) fclose(file);
    (void) close(socket_fd);
    return 2;
// GCOVR_EXCL_STOP
}

int lab_run_transfer(const LabTransferOptions *options)
{
    struct addrinfo hints = {0};
    struct addrinfo *relay = NULL;
    char port[sizeof("65535")];
    int error;
    int status;

    if (options == NULL || options->session == NULL ||
        options->relay == NULL || options->file == NULL) {
        return 2;
    }
    hints.ai_family = AF_UNSPEC;
    hints.ai_socktype = SOCK_DGRAM;
    (void) snprintf(port, sizeof(port), "%u", options->port);
    error = getaddrinfo(options->relay, port, &hints, &relay);
    // GCOVR_EXCL_START
    if (error != 0) {
        fprintf(stderr, "cannot resolve relay %s: %s\n",
                options->relay, gai_strerror(error));
        return 2;
    } // GCOVR_EXCL_STOP
    status = options->mode == LAB_SEND_MODE ? run_sender(options, relay) :
                                               run_receiver(options, relay);
    freeaddrinfo(relay);
    return status;
}

static void usage(FILE *stream)
{
    fprintf(stream,
            "Usage: myapp send -s <session> [-w window] [-T timeout-ms] "
            "[-l loss]\n"
            "              [-c corrupt] [-d dup] [-p port] <relay> <file>\n"
            "       myapp recv -s <session> [-p port] <relay> <file>\n");

static bool unsigned_value(const char *text, unsigned int *value)
{
    char *end = NULL;
    unsigned long parsed;

    if (text == NULL || *text == '\0' || text[0] == '-') {
        return false;
    }
    errno = 0;
    parsed = strtoul(text, &end, 10);
    if (errno == ERANGE || end == text || *end != '\0' || parsed > UINT_MAX) {
        return false;
    }
    *value = (unsigned int) parsed;
    return true;
}

static bool probability_value(const char *text, double *value)
{
    char *end = NULL;
    double parsed;

    if (text == NULL || *text == '\0') {
        return false;
    }
    errno = 0;
    parsed = strtod(text, &end);
    if (errno == ERANGE || end == text || *end != '\0' ||
        !isfinite(parsed) || parsed < 0.0 || parsed > 0.5) {
        return false;
    }
    *value = parsed;
    return true;
}

static bool valid_session(const char *session)
{
    size_t length;

    if (session == NULL) {
        return false;
    }
    length = strlen(session);
    if (length == 0U || length > LAB_MAX_SESSION_LENGTH) {
        return false;
    }
    for (size_t i = 0; i < length; i++) {
        char c = session[i];
        if (!((c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') ||
              c == '-')) {
            return false;
        }
    }
    return true;
}

static bool parse_options(int argc, char **argv, LabTransferOptions *options)
{
    const char *format;
    int option;

    if (argc < 2 || (strcmp(argv[1], "send") != 0 &&
                     strcmp(argv[1], "recv") != 0)) {
        return false;
    }
    options->mode = strcmp(argv[1], "send") == 0 ? LAB_SEND_MODE :
                                                   LAB_RECEIVE_MODE;
    options->session = NULL;
    options->window = LAB_DEFAULT_WINDOW;
    options->timeout_ms = LAB_DEFAULT_TIMEOUT_MS;
    options->port = LAB_DEFAULT_PORT;
    options->loss = 0.0;
    options->corrupt = 0.0;
    options->duplicate = 0.0;
    format = options->mode == LAB_SEND_MODE ? "s:w:T:l:c:d:p:" : "s:p:";
    optind = 2;
    opterr = 0;
    while ((option = getopt(argc, argv, format)) != -1) {
        switch (option) {
        case 's':
            options->session = optarg;
            break;
        case 'w':
            if (!unsigned_value(optarg, &options->window) ||
                options->window < 1U || options->window > LAB_MAX_WINDOW) {
                return false;
            }
            break;
        case 'T':
            if (!unsigned_value(optarg, &options->timeout_ms) ||
                options->timeout_ms == 0U) {
                return false;
            }
            break;
        case 'l':
            if (!probability_value(optarg, &options->loss)) return false;
            break;
        case 'c':
            if (!probability_value(optarg, &options->corrupt)) return false;
            break;
        case 'd':
            if (!probability_value(optarg, &options->duplicate)) return false;
            break;
        case 'p':
            {
                unsigned int port;
                if (!unsigned_value(optarg, &port) || port == 0U ||
                    port > UINT16_MAX) {
                    return false;
                }
                options->port = (uint16_t) port;
            }
            if (options->port == 0U) {
                return false;
            }
            break;
        default:
            return false;
        }
    }
    if (!valid_session(options->session) || argc - optind != 2) {
        return false;
    }
    options->relay = argv[optind];
    options->file = argv[optind + 1];
    return options->relay[0] != '\0' && options->file[0] != '\0';
}

int main(int argc, char **argv)
{
    LabTransferOptions options;

    if (argc == 1) {
        usage(stdout);
        return 0;
    }
    if (!parse_options(argc, argv, &options)) {
        usage(stderr);
        return 1;
    }
    return lab_run_transfer(&options);
}

static void queue_action(LabPacket *queue, size_t *length,
                         const LabAction *action, unsigned int *random_state)
{
    unsigned int roll = rand_r(random_state) % 10U;
    uint8_t wire[LAB_MAX_PACKET];
    size_t wire_length;
    LabPacket packet;

    if (roll < 2U) return;
    TEST_ASSERT_TRUE(lab_encode_packet(&action->packet, wire, sizeof(wire),
                                       &wire_length));
    if (roll == 2U && wire_length > LAB_PACKET_HEADER) {
        wire[LAB_PACKET_HEADER] ^= 1U;
    }
    if (!lab_decode_packet(wire, wire_length, &packet)) return;

static void test_relay_sender(int socket_fd)
{
    struct sockaddr_storage peer;
    socklen_t peer_length = sizeof(peer);
    uint8_t buffer[LAB_MAX_PACKET];
    uint8_t encoded[LAB_MAX_PACKET];
    LabPacket packet;
    LabPacket ack = {0};
    ssize_t length = recvfrom(socket_fd, buffer, sizeof(buffer), 0,
                              (struct sockaddr *) &peer, &peer_length);

    if (length > 0 && (size_t) length < sizeof(buffer) &&
        memcmp(buffer, "HELLO ", 6) == 0) {
        (void) sendto(socket_fd, "OK", 2, 0, (struct sockaddr *) &peer,
                      peer_length);
        (void) sendto(socket_fd, "bad", 3, 0, (struct sockaddr *) &peer,
                      peer_length);
    }
    for (;;) {
        length = recvfrom(socket_fd, buffer, sizeof(buffer), 0,
                          (struct sockaddr *) &peer, &peer_length);
        if (length <= 0 || !lab_decode_packet(buffer, (size_t) length, &packet)) {
            _exit(1);
        }
        ack.type = LAB_ACK;
        ack.seq = packet.seq + 1U;
        size_t encoded_length = 0;
        if (!lab_encode_packet(&ack, encoded, sizeof(encoded), &encoded_length)) {
            _exit(1);
        }
        if (sendto(socket_fd, encoded, encoded_length, 0,
                   (struct sockaddr *) &peer, peer_length) < 0) {
            _exit(1);
        }
        if (packet.type == LAB_FIN) {
            _exit(0);
        }
    }
}

static void test_relay_receiver(int socket_fd, const uint8_t *payload,
                                size_t payload_length)
{
    struct sockaddr_storage peer;
    socklen_t peer_length = sizeof(peer);
    uint8_t buffer[LAB_MAX_PACKET];
    uint8_t encoded[LAB_MAX_PACKET];
    LabPacket packet;

                              (struct sockaddr *) &peer, &peer_length);

    if (length <= 0 || (size_t) length >= sizeof(buffer) ||
        memcmp(buffer, "HELLO ", 6) != 0 ||
        sendto(socket_fd, "OK", 2, 0, (struct sockaddr *) &peer,
               peer_length) < 0) {
        _exit(1);
    }
    (void) sendto(socket_fd, "bad", 3, 0, (struct sockaddr *) &peer,
                  peer_length);
    packet.type = LAB_DATA;
    packet.seq = 0;
    packet.length = (uint16_t) payload_length;
    memcpy(packet.payload, payload, payload_length);
    size_t encoded_length = 0;
    if (!lab_encode_packet(&packet, encoded, sizeof(encoded), &encoded_length) ||
        sendto(socket_fd, encoded, encoded_length, 0,
               (struct sockaddr *) &peer, peer_length) < 0) {
        _exit(1);
    }
    packet.type = LAB_FIN;
    packet.seq = 1;
    packet.length = 0;
    if (!lab_encode_packet(&packet, encoded, sizeof(encoded), &encoded_length) ||
        sendto(socket_fd, encoded, encoded_length, 0,
               (struct sockaddr *) &peer, peer_length) < 0) {
        _exit(1);
    }
    for (unsigned int i = 0; i < 2; i++) {
        length = recvfrom(socket_fd, buffer, sizeof(buffer), 0,
                          (struct sockaddr *) &peer, &peer_length);
        if (length <= 0 || !lab_decode_packet(buffer, (size_t) length, &packet) ||
            packet.type != LAB_ACK) {
            _exit(1);
        }
    }
    _exit(0);
}
